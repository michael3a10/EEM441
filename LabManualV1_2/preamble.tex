% Wrap listings in a light panel consistent with the section boxes.
% Usage:  \begin{codepanel}{caption text}{lststyle} ... \begin{lstlisting} ... \end{lstlisting} ... \end{codepanel}
\newenvironment{codepanel}[2]{%
  % Apply the requested listings style BEFORE opening the tcolorbox group,
  % so listings' internal keyword-class state (classoffset, directive
  % classes, etc.) is fully established ahead of the nested tcolorbox/
  % lstlisting groups -- applying it after \begin{tcolorbox} can leave
  % some class registrations (e.g. C/C++ #directives) not fully active
  % by the time \begin{lstlisting} reads them.
  \lstset{style=#2}%
  \begin{tcolorbox}[
    enhanced, breakable,
    colback=codebg, colframe=panelrule,
    boxrule=0.6pt, arc=1.5pt,
    left=4pt, right=4pt, top=3pt, bottom=3pt,
    title={\small\plexmedium\color{slategray} #1},
    fonttitle=\small, coltitle=slategray,
    colbacktitle=panelbg, boxsep=1pt,
  ]
}{%
  \end{tcolorbox}
}

% ---------- Per-experiment numbering for figures/tables/listings ----------
% expnum tracks "which experiment are we in"; figure/table/listing are
% chained to it so they read as N.1, N.2, ... and reset at each new
% \experimentheader, instead of numbering continuously 1..end of book.
\newcounter{expnum}
\newcounter{listing}
\counterwithin{figure}{expnum}
\counterwithin{table}{expnum}
\counterwithin{listing}{expnum}
\renewcommand{\thelisting}{\theexpnum.\arabic{listing}}

% ---------- Experiment title block ----------
\newcommand{\experimentheader}[4]{%
  % #1 = experiment number, #2 = short title, #3 = full title, #4 = track label (e.g. Control Lab)
  \clearpage
  \setcounter{expnum}{#1}%
  \setcounter{figure}{0}%
  \setcounter{table}{0}%
  \setcounter{listing}{0}%
  \phantomsection
  \addcontentsline{toc}{section}{Experiment #1: #3}
  \markboth{Experiment #1 \textbar\ #4}{Experiment #1 \textbar\ #4}
  \begingroup
  \centering
  \vspace*{0.4em}
  {\plexmedium\color{usmpurple}\Huge Experiment #1\par}
  \vspace{0.2em}
  {\plexmedium\color{inkblack}\LARGE #3\par}
  \vspace{0.8em}
  \endgroup
}
