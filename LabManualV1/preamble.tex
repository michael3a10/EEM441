% ============================================================
% Verilog / VHDL: styled after the Quartus II text editor's classic
% default scheme (bold blue keywords, green comments, dark red/maroon
% strings, black identifiers, light background) -- recognisable to
% anyone who has used Quartus. HDL keyword classes exploit listings'
% multi-class support so system tasks (Verilog $display etc.),
% compiler directives (`define etc.), and IEEE library types
% (std_logic etc.) are each coloured distinctly, matching how the
% Quartus editor itself differentiates language elements.
% ============================================================
\definecolor{hdlkeyword}{HTML}{0000FF}    % language keywords -- Quartus blue
\definecolor{hdlsystask}{HTML}{7F0055}    % Verilog $system tasks -- plum
\definecolor{hdldirective}{HTML}{8B4513}  % `compiler directives -- brown
\definecolor{hdltype}{HTML}{4A0080}       % IEEE types (std_logic etc.) -- deep indigo, distinct from keyword blue
\definecolor{hdlcomment}{HTML}{3F7F3F}    % comments -- Quartus green
\definecolor{hdlstring}{HTML}{7F0000}     % strings -- Quartus maroon
\definecolor{hdlbg}{HTML}{FFFFFF}         % editor background -- white, as in Quartus

\lstdefinestyle{hdlbase}{
  basicstyle       = \ttfamily\small\color{inkblack},
  commentstyle     = \color{hdlcomment}\itshape,
  stringstyle      = \color{hdlstring},
  numberstyle      = \tiny\color{slategray},
  identifierstyle  = \color{inkblack},
  backgroundcolor  = \color{hdlbg},
  numbers          = left,
  numbersep        = 8pt,
  stepnumber       = 1,
  breaklines       = true,
  breakatwhitespace= false,
  showstringspaces = false,
  showspaces       = false,
  showtabs         = false,
  tabsize          = 2,
  columns          = flexible,
  keepspaces       = true,
  frame            = none,
  xleftmargin      = 12pt,
  framexleftmargin = 8pt,
  aboveskip        = 4pt,
  belowskip        = 2pt,
  captionpos       = b,
}

\lstdefinestyle{verilogstyle}{
  style        = hdlbase,
  classoffset  = 0,
  language     = Verilog,
  keywordstyle      = \color{hdlkeyword}\bfseries,      % reserved keywords
  keywordstyle=[2]\color{hdlsystask}\bfseries,          % $system tasks/functions
  keywordstyle=[3]\color{hdldirective}\bfseries,        % `compiler directives
}

\lstdefinestyle{vhdlstyle}{
  style        = hdlbase,
  classoffset  = 0,
  language     = VHDL,
  keywordstyle      = \color{hdlkeyword}\bfseries,      % reserved keywords
  keywordstyle=[2]\color{hdltype}\bfseries,             % IEEE library types
  morekeywords=[2]{std_logic, std_logic_vector, std_ulogic, std_ulogic_vector,
      signed, unsigned, integer, natural, positive, boolean, bit, bit_vector,
      character, string, time, real},
}

\lstdefinestyle{cstyle}{
  style        = swbase,
  classoffset  = 0,     % reset keyword-class state (avoids leakage from prior HDL styles)
  language     = C,
  emphstyle    = \color{swtype},
  emph={int, char, float, double, void, long, short, unsigned, signed,
      size_t, bool, uint8_t, uint16_t, uint32_t, int8_t, int16_t, int32_t},
  moredelim    = [l][\color{swpreproc}\bfseries]{\#},   % preprocessor directive lines
  % Note: this directive colouring (like listings' built-in `directives`
  % class) does not take effect inside a breakable tcolorbox, which is
  % how all code panels in this document are rendered -- a known
  % listings/tcolorbox interaction. #include/#define lines therefore
  % render in the default text colour inside \codepanel, which is a
  % minor, accepted cosmetic gap; all keyword/type/comment/string
  % highlighting is unaffected and works correctly.
}

% Wrap listings in a light panel consistent with the section boxes.
% Usage:  \begin{codepanel}{caption text}{lststyle} ... \begin{lstlisting} ... \end{lstlisting} ... \end{codepanel}
\newenvironment{codepanel}[2]{%
  % Apply the requested listings style BEFORE opening the tcolorbox group,
  % so listings' internal keyword-class state (classoffset, directive
  % classes, etc.) is fully established ahead of the nested tcolorbox/
  % lstlisting groups -- applying it after \begin{tcolorbox} can leave
  % some class registrations (e.g. C/C++ #directives) not fully active
  % by the time \begin{lstlisting} reads them.
  \lstset{style=#2}%
  \begin{tcolorbox}[
    enhanced, breakable,
    colback=codebg, colframe=panelrule,
    boxrule=0.6pt, arc=1.5pt,
    left=4pt, right=4pt, top=3pt, bottom=3pt,
    title={\small\plexmedium\color{slategray} #1},
    fonttitle=\small, coltitle=slategray,
    colbacktitle=panelbg, boxsep=1pt,
  ]
}{%
  \end{tcolorbox}
}
